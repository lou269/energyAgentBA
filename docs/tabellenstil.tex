\newenvironment{tabtext}[1]
  {\tabgroesse\tabularx{\linewidth}{@{}#1@{}}\toprule}
  {\bottomrule\endtabularx}

\newenvironment{tabkompakt}[1]
  {\tabgroesse\tabular{@{}#1@{}}\toprule}
  {\bottomrule\endtabular}

\newenvironment{tabzahlen}[1]
  {\tabgroesse\csname tabular*\endcsname{\linewidth}{@{\extracolsep{\fill}}#1@{}}\toprule}
  {\bottomrule\csname endtabular*\endcsname}

\newenvironment{kriterien}
  {\tabgroesse\tabularx{\linewidth}{@{} L P{2.6cm} P{4.2cm} M @{}}
   \toprule Kriterium & Ziel & Ergebnis & erfüllt \\ \midrule}
  {\bottomrule\endtabularx}

\newenvironment{konfig}[1]
  {\small\longtable{@{} K{0.62\linewidth} K{\dimexpr0.38\linewidth-2\tabcolsep\relax} @{}}
   \caption{Konfiguration \texttt{#1}}\label{tab:cfg-#1}\\
   \toprule Parameter & Wert \\ \midrule
   \endfirsthead
   \multicolumn{2}{@{}l}{\normalfont\textit{Fortsetzung von Tab.~\thetable}}\\
   \toprule Parameter & Wert \\ \midrule
   \endhead
   \bottomrule
   \endlastfoot}
  {\endlongtable}

\newcommand{\tabkopflinie}{\noalign{\global\tabkopftrue}\tabOrigToprule}
\newenvironment{tablang}[3][\tabgroesse]
  {#1\xltabular{\linewidth}{@{}#2@{}}\toprule #3 \\ \midrule \endfirsthead
   \tabkopflinie #3 \\ \midrule \endhead}
  {\bottomrule\endxltabular}

\newenvironment{konfigvar}[4]
  {\small\longtable{@{} K{0.30\linewidth} *{#3}{C} @{}}
   \caption{#1}\label{#2}\\
   \toprule #4 \\ \midrule
   \endfirsthead
   \multicolumn{\numexpr#3+1\relax}{@{}l}{\normalfont\textit{Fortsetzung von Tab.~\thetable}}\\
   \toprule #4 \\ \midrule
   \endhead
   \bottomrule
   \endlastfoot}
  {\endlongtable}
